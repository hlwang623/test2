from sqlalchemy.orm import Session
from app.models.message import Message
from typing import List, Optional
from datetime import datetime

def create_message(
        db: Session,
        user_id: int,
        type: str,
        title: str,
        sender: str,
        content: str,
        icon_color: str = "bg-blue-100 text-blue-500",
        related_subject_id: Optional[int] = None,
        related_file_id: Optional[int] = None
) -> Message:
    """创建新消息"""
    message = Message(
        user_id=user_id,
        type=type,
        title=title,
        sender=sender,
        content=content,
        icon_color=icon_color,
        related_subject_id=related_subject_id,
        related_file_id=related_file_id
    )
    db.add(message)
    db.commit()
    db.refresh(message)
    return message


def get_user_messages(db: Session, user_id: int, skip: int = 0, limit: int = 100) -> List[Message]:
    """获取用户的所有消息（最新的在前）"""
    return db.query(Message) \
        .filter(Message.user_id == user_id) \
        .order_by(Message.created_at.desc()) \
        .offset(skip) \
        .limit(limit) \
        .all()


def get_unread_count(db: Session, user_id: int) -> int:
    """获取未读消息数量"""
    return db.query(Message) \
        .filter(Message.user_id == user_id, Message.is_read == False) \
        .count()


def mark_as_read(db: Session, message_id: int, user_id: int) -> Optional[Message]:
    """标记消息为已读"""
    message = db.query(Message) \
        .filter(Message.id == message_id, Message.user_id == user_id) \
        .first()

    if message:
        message.is_read = True
        db.commit()
        db.refresh(message)

    return message


def mark_all_as_read(db: Session, user_id: int) -> int:
    """标记所有消息为已读"""
    count = db.query(Message) \
        .filter(Message.user_id == user_id, Message.is_read == False) \
        .update({"is_read": True})
    db.commit()
    return count


def delete_message(db: Session, message_id: int, user_id: int) -> bool:
    """删除消息"""
    message = db.query(Message) \
        .filter(Message.id == message_id, Message.user_id == user_id) \
        .first()

    if message:
        db.delete(message)
        db.commit()
        return True

    return False
