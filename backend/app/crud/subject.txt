import json
import logging
import random
import os
from datetime import datetime, timezone
from typing import Any, Dict, Union, List, Optional
from sqlalchemy.orm import Session
from pydantic import BaseModel
from app.models.subject import Subject, SubjectFile, FileStatus
from app.schemas.subject import SubjectCreate
from app.utils.toc_extractor import extract_toc_from_parsed_files
# 配置 Logger
logger = logging.getLogger(__name__)
# 封面颜色池
COVER_COLORS = ["blue", "orange", "purple", "green", "pink", "teal"]
def get_subject(db: Session, subject_id: int) -> Optional[Subject]:
    return db.query(Subject).filter(Subject.id == subject_id).first()

def get_user_subjects(db: Session, user_id: int, skip: int = 0, limit: int = 100) -> List[Subject]:
    return db.query(Subject) \
        .filter(Subject.user_id == user_id) \
        .order_by(Subject.updated_at.desc()) \
        .offset(skip) \
        .limit(limit) \
        .all()


def get_all_subjects(db: Session) -> List[Subject]:
    """管理员获取所有用户的所有科目"""
    return db.query(Subject).order_by(Subject.updated_at.desc()).all()


def get_subjects_by_user(db: Session, user_id: int) -> List[Subject]:
    """管理员获取指定用户的所有科目"""
    return db.query(Subject).filter(Subject.user_id == user_id).order_by(Subject.updated_at.desc()).all()


def create_subject(db: Session, subject: SubjectCreate, user_id: int) -> Subject:
    db_subject = Subject(
        name=subject.name,
        description=subject.description,
        user_id=user_id,
        cover_color=random.choice(COVER_COLORS),
        total_nodes=0,
        learned_nodes=0,
        status="empty"
    )
    db.add(db_subject)
    db.commit()
    db.refresh(db_subject)
    return db_subject


def update_subject(
        db: Session,
        db_obj: Subject,
        obj_in: Union[Dict[str, Any], BaseModel]
) -> Subject:
    """
    通用更新函数
    """
    if isinstance(obj_in, dict):
        update_data = obj_in
    else:
        # 显式调用 Pydantic V2 的 model_dump
        update_data = obj_in.model_dump(exclude_unset=True)

    for field, value in update_data.items():
        if hasattr(db_obj, field):
            setattr(db_obj, field, value)

    # 使用 timezone-aware UTC 时间
    db_obj.updated_at = datetime.now(timezone.utc)
    db.add(db_obj)
    db.commit()
    db.refresh(db_obj)
    return db_obj


def get_file(db: Session, file_id: int) -> Optional[SubjectFile]:
    return db.query(SubjectFile).filter(SubjectFile.id == file_id).first()


def update_file_status(
        db: Session,
        file_id: int,
        status: FileStatus = None,
        parsed_content_path: str = None,
        parsed_json_path: str = None,
        parsing_error: str = None,
        parsed_at: datetime = None,
        parse_progress: float = None
) -> Optional[SubjectFile]:
    """
    更新文件的解析状态
    用于 MinerU 后台任务更新进度
    """
    db_file = get_file(db, file_id)
    if not db_file:
        return None

    # 只更新非 None 的字段
    if status is not None:
        db_file.status = status
    if parsed_content_path is not None:
        db_file.parsed_content_path = parsed_content_path
    if parsed_json_path is not None:
        db_file.parsed_json_path = parsed_json_path
    if parsing_error is not None:
        db_file.parsing_error = parsing_error
    if parsed_at is not None:
        db_file.parsed_at = parsed_at
    if parse_progress is not None:
        db_file.parse_progress = parse_progress

    db_file.updated_at = datetime.now(timezone.utc)
    db.add(db_file)
    db.commit()
    db.refresh(db_file)
    return db_file


def delete_file(db: Session, file_id: int) -> Optional[SubjectFile]:
    """软删除文件"""
    db_file = get_file(db, file_id)
    if db_file:
        db_file.is_deleted = True
        db_file.deleted_at = datetime.now(timezone.utc)
        db.add(db_file)
        db.commit()
        db.refresh(db_file)
        # 删除文件后可能需要更新科目状态
        if db_file.subject_id:
            update_subject_status(db, db_file.subject_id)
    return db_file


def delete_subject(db: Session, subject_id: int) -> None:
    db_obj = get_subject(db, subject_id)
    if db_obj:
        db.delete(db_obj)
        db.commit()


def add_file_to_subject(
        db: Session,
        subject_id: int,
        filename: str,  # 带时间戳的存储文件名
        original_filename: str,
        file_path: str,
        pdf_path: str = None,
        conversion_status: str = 'not_needed',
        file_type: str = None,
        file_size: int = 0
) -> SubjectFile:
    db_file = SubjectFile(
        subject_id=subject_id,
        file_name=filename,
        original_filename=original_filename,
        file_path=file_path,
        pdf_path=pdf_path,
        conversion_status=conversion_status,
        file_type=file_type,
        file_size=file_size,
        status=FileStatus.PENDING,
        graph_status="pending",
        parse_progress=0.0
    )
    db.add(db_file)

    subject = get_subject(db, subject_id)
    if subject:
        subject.updated_at = datetime.now(timezone.utc)
        db.add(subject)

    db.commit()
    return db_file


# --- 状态管理逻辑 ---

def update_file_graph_status(db: Session, file_id: int, status: str) -> Optional[SubjectFile]:
    """更新单个文件的图谱状态"""
    db_file = get_file(db, file_id)
    if db_file:
        db_file.graph_status = status
        db.add(db_file)
        db.commit()
        db.refresh(db_file)
    return db_file


def update_subject_status(db: Session, subject_id: int) -> Optional[Subject]:
    """
    根据所有文件的 graph_status 计算科目的 status
    """
    subject = get_subject(db, subject_id)
    if not subject:
        return None

    # 重新加载 files 关系以确保数据最新
    db.refresh(subject)

    has_building = any(f.graph_status == 'building' for f in subject.files)
    has_built = any(f.graph_status == 'built' for f in subject.files)

    new_status = 'empty'
    if has_building:
        new_status = 'parsing'
    elif has_built:
        new_status = 'ready'
    # 否则保持 empty (全都是 pending 或 excluded)

    if subject.status != new_status:
        subject.status = new_status
        db.add(subject)
        db.commit()
        db.refresh(subject)

    return subject


def start_graph_build(db: Session, subject_id: int, file_ids: List[int]) -> Optional[Subject]:
    """
    批量开始构建
    """
    # 1. 将指定文件的 graph_status 改为 'building'
    # 使用 synchronize_session=False 以提高批量更新性能
    db.query(SubjectFile).filter(
        SubjectFile.id.in_(file_ids),
        SubjectFile.subject_id == subject_id
    ).update({SubjectFile.graph_status: "building"}, synchronize_session=False)

    db.commit()

    # 2. 触发后台任务 (MinerU + GraphRAG)
    print(f"[System] Triggered graph build for subject {subject_id}, files: {file_ids}")

    # 3. 更新科目整体状态
    update_subject_status(db, subject_id)

    return get_subject(db, subject_id)


def get_trash_files(db: Session, subject_id: int) -> List[SubjectFile]:
    """获取科目的回收站文件"""
    return db.query(SubjectFile).filter(
        SubjectFile.subject_id == subject_id,
        SubjectFile.is_deleted == True
    ).order_by(SubjectFile.deleted_at.desc()).all()


def restore_file(db: Session, file_id: int) -> Optional[SubjectFile]:
    """还原已删除的文件"""
    db_file = get_file(db, file_id)
    if db_file and db_file.is_deleted:
        db_file.is_deleted = False
        db_file.deleted_at = None
        db.add(db_file)
        db.commit()
        db.refresh(db_file)
        # 还原后更新科目状态
        if db_file.subject_id:
            update_subject_status(db, db_file.subject_id)
    return db_file


def permanent_delete_file(db: Session, file_id: int) -> bool:
    """永久删除文件（物理删除）"""
    db_file: Optional[SubjectFile] = get_file(db, file_id)
    if db_file:
        # 物理删除文件
        # 类型检查修复：确保 db_file.file_path 是字符串
        if db_file.file_path and os.path.exists(db_file.file_path):
            try:
                os.remove(db_file.file_path)
            except Exception as e:
                print(f"警告：物理文件删除失败 - {str(e)}")

        # 删除PDF转换文件
        if db_file.pdf_path and os.path.exists(db_file.pdf_path):
            try:
                os.remove(db_file.pdf_path)
            except Exception:
                pass

        # 数据库删除
        db.delete(db_file)
        db.commit()
        return True
    return False


def get_file_toc(db: Session, subject_id: int, file_id: int) -> Optional[List[Dict[str, Any]]]:
    """
    获取文件的目录结构 (带缓存机制)

    Args:
      db: 数据库会话
      subject_id: 科目ID
      file_id: 文件ID

    Returns:
      list[dict]: 目录列表，如果文件不存在返回 None

    Raises:
      ValueError: 如果文件未解析完成
    """
    # 1. 查询文件
    file_record = db.query(SubjectFile).filter(
        SubjectFile.id == file_id,
        SubjectFile.subject_id == subject_id,
        SubjectFile.is_deleted == False
    ).first()

    if not file_record:
        return None

    if file_record.status != FileStatus.READY:
        raise ValueError(f"文件尚未解析完成，当前状态: {file_record.status.value}")

    # 2. 检查缓存
    if getattr(file_record, 'toc_json', None):
        try:
            toc = json.loads(file_record.toc_json)
            logger.info(f"从缓存读取目录: subject={subject_id}, file={file_id}, 共{len(toc)}项")
            return toc
        except json.JSONDecodeError:
            logger.warning(f"缓存目录解析失败，将重新生成: file_id={file_id}")
            pass

    # 3. 首次生成 (或缓存失效)
    # 调用提取函数 (toc_extractor 内部会默认尝试使用 LLM)
    logger.info(f"首次生成目录(调用LLM): file_id={file_id}")
    try:
        toc = extract_toc_from_parsed_files(subject_id, file_id)
    except Exception as e:
        logger.error(f"生成目录时发生错误: {e}")
        # 如果生成失败，返回空列表而不是报错，避免前端崩溃
        return []

    # 4. 存入数据库缓存
    try:
        if hasattr(file_record, 'toc_json'):
            file_record.toc_json = json.dumps(toc, ensure_ascii=False)
            if hasattr(file_record, 'toc_generated_at'):
                file_record.toc_generated_at = datetime.now(timezone.utc)

            db.add(file_record)
            db.commit()
            logger.info(f"目录已缓存到数据库: file_id={file_id}")
    except Exception as e:
        logger.error(f"保存目录缓存失败: {e}")
        pass

    return toc
