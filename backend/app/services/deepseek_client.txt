import os
import json
import logging
from typing import List, Dict, Any, Optional
from openai import OpenAI

# 尝试导入项目配置，如果不存在则使用环境变量作为降级方案
try:
    from config import DEEPSEEK_API_KEY
    API_KEY = DEEPSEEK_API_KEY
except ImportError:
    # 开发环境/测试环境回退
    API_KEY = os.getenv("DEEPSEEK_API_KEY", "")

# Deepseek 官方 Base URL
DEEPSEEK_BASE_URL = "https://api.deepseek.com"

logger = logging.getLogger(__name__)


class DeepseekClient:
    def __init__(self):
        """
        初始化 Deepseek 客户端
        """
        if not API_KEY:
            logger.warning("Deepseek API Key 未配置，智能目录清洗功能将不可用。")

        # 使用 OpenAI SDK 初始化 (Deepseek 兼容 OpenAI 协议)
        self.client = OpenAI(
            api_key=API_KEY,
            base_url=DEEPSEEK_BASE_URL,
            timeout=120.0,  # 设置超时时间为 30 秒
            max_retries=2  # 自动重试 2 次
        )

    def clean_toc(self, raw_items: List[Dict[str, Any]], file_type: str = "pdf") -> Optional[List[Dict[str, Any]]]:
        """
        利用 LLM 清洗 MinerU 提取的粗糙目录结构

        Args:
            raw_items: 包含 text, page_idx 等信息的原始列表
            file_type: "pdf" 或 "pptx" (不同类型使用不同 Prompt)

        Returns:
            List[Dict]: 清洗后的目录结构 [{"title":Str, "level":Int, "original_text":Str}, ...]
            None: 如果 API 调用失败或解析失败
        """
        if not API_KEY:
            return None

        # 1. 准备数据：只发送必要的字段以节省 Token
        # 我们只需要 text 及其上下文，不需要具体的坐标信息
        simplified_items = [
            {"id": idx, "text": item.get("text", "").strip()}
            for idx, item in enumerate(raw_items)
            if item.get("text", "").strip()
        ]

        if not simplified_items:
            return []

        raw_items_json = json.dumps(simplified_items, ensure_ascii=False, indent=2)

        # 2. 构建 Prompt
        system_prompt = (
            "你是一个专业的文档编辑助手，擅长分析学术论文、技术报告和演示文稿的结构。"
            "你的任务是从粗糙的文本提取中生成干净、层级分明的目录。"
        )

        user_prompt = self._build_user_prompt(file_type, raw_items_json)

        try:
            # 3. 调用 API
            response = self.client.chat.completions.create(
                model="deepseek-chat",  # 使用 V3 或最新模型
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],
                # 强制输出 JSON，极大提高稳定性
                response_format={"type": "json_object"},
                temperature=0.1,  # 低温度，保证结果确定性
            )

            # 4. 解析结果
            content = response.choices[0].message.content
            if not content:
                logger.error("Deepseek 返回内容为空")
                return None

            result = json.loads(content)
            toc = result.get("toc", [])

            logger.info(f"Deepseek 成功清洗目录，生成 {len(toc)} 条目录项")
            return toc

        except json.JSONDecodeError:
            logger.error("Deepseek 返回的不是有效的 JSON 格式")
            return None
        except Exception as e:
            logger.error(f"Deepseek API 调用失败: {str(e)}")
            return None

    def extract_toc_from_text(self, toc_raw_text: str) -> Optional[List[Dict[str, Any]]]:
        """
        新方法：直接从目录页原始文本中提取结构化目录（含逻辑页码）。

        专为 MinerU OCR 错误设计：
        - § 可能被识别为 $\\S\\O1$、81、SI 等
        - 多个目录项可能合并为一个文本块
        - 页码与点线之间可能有噪音

        Returns:
            List[Dict]: [{"title": str, "level": int, "logical_page": int}, ...]
            None: API调用失败
        """
        if not API_KEY:
            logger.warning("Deepseek API Key 未配置，extract_toc_from_text 不可用")
            return None

        logger.info(f"=== [DeepSeek TOC] 开始调用，输入文本 {len(toc_raw_text)} 字符 ===")
        logger.debug(f"  发送文本预览（前200字）:\n{toc_raw_text[:200]}")

        system_prompt = (
            "你是专业的中文PDF文档分析专家，"
            "擅长从OCR识别的目录页文本中提取结构化信息，"
            "并能自动识别和纠正常见OCR错误。"
            "你必须直接输出JSON，不要包含任何解释性文字、开场白或结束语。"
        )

        user_prompt = f"""以下是从中文教材PDF目录页通过OCR识别的原始文本。

【重要】请自动纠正以下常见OCR错误：
1. 章节符号"§"可能被误识别为："$\\S\\O1$"、"81"、"SI"、"S1"、"$S$"等形式
2. 数字和文字之间可能有多余空格（如"第 1 章"应为"第1章"）
3. 目录中多个条目可能被合并为一行（用空格隔开，需要拆分）
4. 每行末尾的页码数字（如"...........15"中的15）是逻辑页码，请务必提取

目录原始文本：
---
{toc_raw_text}
---

任务：
1. 从文本中提取所有目录条目（包括章、节、小节）
2. 纠正OCR错误，还原正确的标题文本（如将"81.1"还原为"§1.1"）
3. 判断层级（按以下规则，优先使用编号规则，无编号时使用相对位置规则）：

   【编号规则】
   - "第X章" / "Chapter X" / 无编号的独立大标题 → level 1
   - "§X.Y" / "第X.Y节" / "X.Y 标题" → level 2
   - "§X.Y.Z" / "X.Y.Z 标题" → level 3

   【相对位置规则（无明显编号时）】
   - 章末附加项：出现在某章最后几个条目中的"小结"、"习题"、"思考题"、"本章小结"、"章末习题"、"参考文献"（紧跟在具体章后） → level 2
   - 书末附加项：出现在所有章之后的独立条目，如"附录"、"附录A/B/C"、"后记"、"索引"、"参考文献"（全书级别） → level 1
   - 节内特殊项：出现在某节内部的"例题"、"扩展阅读"、"注记"、"历史注记" → level 3

4. 提取每条目末尾的页码数字（即书中印刷的逻辑页码）

严格按以下JSON格式输出，不含任何额外文字：
{{
  "toc": [
    {{"title": "第一章 概率论的基本概念", "level": 1, "logical_page": 1}},
    {{"title": "§1.1 随机试验", "level": 2, "logical_page": 3}},
    {{"title": "§1.2 样本空间与随机事件", "level": 2, "logical_page": 7}},
    {{"title": "小结", "level": 2, "logical_page": 45}},
    {{"title": "习题一", "level": 2, "logical_page": 47}},
    {{"title": "附录", "level": 1, "logical_page": 210}}
  ]
}}"""

        try:
            response = self.client.chat.completions.create(
                model="deepseek-chat",
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],
                response_format={"type": "json_object"},
                temperature=0,  # 信息抽取任务无需发散，0温度结果最确定、速度最快
            )

            # 记录 Token 消耗
            usage = response.usage
            if usage:
                logger.info(
                    f"  Token消耗: prompt={usage.prompt_tokens}, "
                    f"completion={usage.completion_tokens}, "
                    f"total={usage.total_tokens}"
                )

            content = response.choices[0].message.content
            if not content:
                logger.error("  Deepseek 返回内容为空")
                return None

            logger.debug(f"  LLM原始响应:\n{content}")

            result = json.loads(content)
            toc = result.get("toc", [])

            # 逐条打印提取结果
            for i, item in enumerate(toc):
                logger.debug(
                    f"  [{i + 1}] level={item.get('level')} "
                    f"logical_page={item.get('logical_page')} "
                    f"title='{item.get('title', '')[:50]}'"
                )

            if toc:
                logger.info(f"=== [DeepSeek TOC] 提取完成，共 {len(toc)} 条 ===")
            else:
                logger.warning(
                    f"=== [DeepSeek TOC] 模型返回空目录！"
                    f"输入文本长度={len(toc_raw_text)} 字符，"
                    f"预览: {toc_raw_text[:200]!r} ==="
                )
            return toc

        except json.JSONDecodeError:
            logger.error("  Deepseek 返回的不是有效 JSON 格式")
            return None
        except Exception as e:
            logger.error(f"  Deepseek API 调用失败: {str(e)}")
            return None

    def _build_user_prompt(self, file_type: str, raw_items_json: str) -> str:
        """根据文件类型构建特定的 User Prompt"""

        base_req = """
        输出格式：严格按照以下JSON格式输出，不要有任何其他文字：
        {
          "toc": [
            {
              "title": "摘要",
              "level": 1,
              "original_text": "摘要"
            },
            {
              "title": "1. 信源建模",
              "level": 2,
              "original_text": "1. 信源建模：交互历史的随机过程化"
            }
          ]
        }
        要求：
        - title: 清洗后的标题文本（去除多余的页码点号，保留章节编号）
        - level: 1-3，1是一级标题，2是二级，3是三级
        - original_text: 必须严格等于原始数据中的 text 字段（用于后续我程序里的精确匹配回填页码）
        """

        if file_type == "pptx":
            return f"""
            这是一份PPT演示文稿的文本提取。PPT的特点：
            - 每一页通常有一个标题
            - 有些是"章节页"（单独大标题），有些是"内容页"
            - 需要你归纳出逻辑层级

            任务：
            1. 识别哪些是章节页（Part 1, 第一章等），设为 Level 1
            2. 将内容页归类到对应章节下，设为 Level 2
            3. 剔除封面信息（演讲者、日期、感谢观看等）
            4. 如果没有明显的章节页，请根据内容逻辑归纳结构

            原始数据：
            {raw_items_json}

            {base_req}
            """
        else:
            # 默认为 PDF 处理逻辑
            return f"""
            我给你一份从PDF文档中提取的文本片段列表，这些片段来自文档的前几页和所有标题级别的文本。列表中包含：
            - 封面信息（学校名、作者、日期等）
            - 目录页本身
            - 正文标题
            - 页眉页脚等噪音

            任务：
            1. 剔除所有非内容的信息（封面、作者信息、"目录"这个标题本身、页眉页脚）
            2. 识别文档的真正目录结构
            3. 如果文档有目录页，优先以目录页为准
            4. 如果没有目录页，根据标题总结出逻辑目录

            原始数据：
            {raw_items_json}

            {base_req}
            """


# 单例模式导出（可选）
deepseek_client = DeepseekClient()
