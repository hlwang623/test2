@router.get("/{subject_id}", response_model=SubjectResponse)
def get_subject(
        subject_id: int,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
):
    subject = crud_subject.get_subject(db, subject_id=subject_id)
    if not subject or subject.user_id != current_user.id:
        raise HTTPException(status_code=404, detail="科目不存在或无权查看")

    subject.files = [f for f in subject.files if not f.is_deleted]

    return subject

@router.delete("/{subject_id}")
def delete_subject(
        subject_id: int,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
):
    subject = crud_subject.get_subject(db, subject_id=subject_id)
    if not subject or subject.user_id != current_user.id:
        raise HTTPException(status_code=404, detail="科目不存在")

    # 物理删除
    for file_record in subject.files:
        if os.path.exists(file_record.file_path):
            try:
                os.remove(file_record.file_path)
            except:
                pass

    crud_subject.delete_subject(db, subject_id=subject_id)
    return {"message": "删除成功"}

@router.post("/{subject_id}/files/upload_only", response_model=SubjectResponse)
async def upload_file_only(
        subject_id: int,
        background_tasks: BackgroundTasks,
        file: UploadFile = File(...),
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
):
    subject = crud_subject.get_subject(db, subject_id=subject_id)
    if not subject or subject.user_id != current_user.id:
        raise HTTPException(status_code=403, detail="无权操作")

    if file.content_type not in ALLOWED_MIME_TYPES:
        raise HTTPException(status_code=400, detail="不支持的文件类型")

    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    safe_filename = f"{subject_id}_{timestamp}_{file.filename.replace(' ', '_')}"
    file_path = os.path.join(UPLOAD_DIR, safe_filename)

    try:
        with open(file_path, "wb") as buffer:
            content = await file.read()
            if len(content) > MAX_FILE_SIZE:
                raise HTTPException(status_code=413, detail=f"文件过大，最大支持 {MAX_FILE_SIZE // (1024 * 1024)}MB")

            buffer.write(content)
        file_size = os.path.getsize(file_path)
        file_ext = file.filename.split('.')[-1].lower()
        needs_conversion = file_ext in ['docx', 'doc', 'pptx', 'ppt']

        pdf_path = None
        conversion_status = 'not_needed'

        if needs_conversion:
            conversion_status = 'pending'  # 标记为待转换
            pdf_path = None  # 后台任务会填充这个值
        crud_subject.add_file_to_subject(
            db=db,
            subject_id=subject_id,
            filename=safe_filename,  # 存储文件名（带时间戳）
            original_filename=file.filename,
            file_path=file_path,
            pdf_path=pdf_path,
            conversion_status=conversion_status,
            file_type=file.content_type,
            file_size=file_size
        )
        # 确定要解析的PDF路径
        final_pdf_path = file_path
        # 获取刚创建的文件记录
        db.refresh(subject)  # 刷新subject以获取最新的files
        file_record = subject.files[-1]  # 获取最后一个文件
        background_tasks.add_task(
            process_pdf_parsing,
            subject_id=subject_id,
            file_id=file_record.id,
            pdf_path=final_pdf_path
        )
    except Exception as e:
        if os.path.exists(file_path): os.remove(file_path)
        raise HTTPException(status_code=500, detail=str(e))

    db.refresh(subject)
    return subject

@router.post("/{subject_id}/files/{file_id}/regenerate_toc", response_model=TocResponse)
async def regenerate_file_toc(
        subject_id: int,
        file_id: int,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
):
    """
    手动触发重新生成文件目录（用于调试或更新旧数据）
    """
    import json
    from datetime import datetime, timezone
    from app.utils.toc_extractor import extract_toc_from_parsed_files

    # 1. 验证权限
    subject = crud_subject.get_subject(db, subject_id=subject_id)
    if not subject or subject.user_id != current_user.id:
        raise HTTPException(status_code=403, detail="无权操作")

    # 2. 验证文件存在且已解析完成
    file_record = crud_subject.get_file(db, file_id=file_id)
    if not file_record or file_record.subject_id != subject_id:
        raise HTTPException(status_code=404, detail="文件不存在")
    if file_record.status.value != "ready":
        raise HTTPException(status_code=400, detail="文件尚未解析完成，无法生成目录")

    # 3. 重新提取目录（双轨制：优先原生书签，降级到 LLM）
    try:
        import pypdf
        from app.utils.toc_extractor import extract_toc_from_pdf_outline

        # 取实际 PDF 路径：优先用格式转换后的 pdf_path，原文件已是 PDF 则用 file_path
        actual_pdf_path = file_record.pdf_path if file_record.pdf_path else file_record.file_path

        try:
            total_pages = len(pypdf.PdfReader(actual_pdf_path).pages)
        except Exception:
            total_pages = 9999  # 读取失败时不阻断流程，越界校验放宽

        native_toc = extract_toc_from_pdf_outline(actual_pdf_path, total_pages)

        if native_toc:
            toc = native_toc
        else:
            toc = extract_toc_from_parsed_files(subject_id, file_id)
    except Exception as e:
        raise HTTPException(status_code=500, detail=f"目录提取失败: {str(e)}")

    # 4. 写回数据库
    file_record.toc_json = json.dumps(toc, ensure_ascii=False)
    file_record.toc_generated_at = datetime.now(timezone.utc)
    db.add(file_record)
    db.commit()

    return {
        "toc": toc,
        "total": len(toc),
        "message": f"目录已重新生成，共 {len(toc)} 项"
    }


@router.post("/{subject_id}/files/{file_id}/ocr")
async def ocr_region(
        subject_id: int,
        file_id: int,
        payload: dict,
        current_user=Depends(get_current_user),
        db: Session = Depends(get_db)
):
    """区域框选 OCR：接收前端截取的 Base64 PNG，返回识别文字"""
    from app.services.ocr_service import ocr_from_base64
    image_base64 = payload.get("image_base64", "")
    if not image_base64:
        raise HTTPException(status_code=400, detail="image_base64 不能为空")
    text = ocr_from_base64(image_base64)
    return {"text": text}


@router.get("/{subject_id}/files/{file_id}/toc", response_model=TocResponse)
async def get_file_table_of_contents(
        subject_id: int,
        file_id: int,
        current_user: User = Depends(get_current_user),
        db: Session = Depends(get_db)
):
    """
    获取文件的目录结构

    Args:
        subject_id: 科目ID
        file_id: 文件ID

    Returns:
        TocResponse: 包含目录列表的响应

    Raises:
        403: 无权访问
        404: 文件不存在
        400: 文件未解析完成
    """
    # 1. 验证科目权限
    subject = crud_subject.get_subject(db, subject_id=subject_id)
    if not subject or subject.user_id != current_user.id:
        raise HTTPException(status_code=403, detail="无权访问")

    # 2. 获取TOC (用 try-except)
    try:
        toc = crud_subject.get_file_toc(db, subject_id, file_id)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

    # 3. 边界情况处理
    if toc is None:
        raise HTTPException(status_code=404, detail="文件不存在")

    # 4. 空目录处理
    if len(toc) == 0:
        return {
            "toc": [],
            "total": 0,
            "message": "该文件未检测到目录结构"
        }

    # 5. 正常返回
    return {
        "toc": toc,
        "total": len(toc)
    }
